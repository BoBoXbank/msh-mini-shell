# ==========================================================================
#  demo-steps.txt — the commands ./demo.sh types for you, in this order
#
#  In ./demo.sh:  Enter = type the next command,  Enter again = run it.
#  Lines that start with # are notes and are skipped.
#  "@wait N" = before typing the next command, make sure N seconds have
#  passed since the previous one was run.
# ==========================================================================

# 1 · a normal pipeline (§4.3)
ls -l | wc -l

# 2-3 · x-ray on, then the line from the roadmap slide (§2.1–§5.1)
export MSH_TRACE=1
ls -l | wc -l > out.txt &

# 4 · x-ray off; the [done] lines come from the SIGCHLD handler (§5.2)
export MSH_TRACE=0

# 5 · press Ctrl-C while it sleeps: the job dies, the shell survives (§5.4)
sleep 30

# 6-8 · the hidden hang (§4.5): press Ctrl-C when it hangs
export MSH_LEAK=1
ls | wc -l
export MSH_LEAK=0

# 9-12 · a zombie (§5.1), then the reaper cleans it up (§5.2)
export MSH_REAPER=0
sleep 1 &
@wait 1.3
ps -o pid,stat,cmd
export MSH_REAPER=1

# 13 · the end
exit
